3NF NORMALIZED RELATIONAL MODEL
SATELLITE Core Entity
  • PK Satellite_ID : INT
  • Satellite_Name : VARCHAR(100)
  • Satellite_Type : VARCHAR(50)
  • Manufacturer : VARCHAR(100)
  • Launch_Date : DATE
  • Mass : DECIMAL(10,2)
  • Status : VARCHAR(20)
  • FK Orbit_ID : INT (→ ORBIT)
ORBIT 1 : N Satellites
  • PK Orbit_ID : INT
  • Orbit_Type : VARCHAR(50)
  • Altitude : DECIMAL(10,2) (km)
  • Inclination : DECIMAL(6,2) (deg)
  • Eccentricity : DECIMAL(8,5)
  • Orbital_Period : DECIMAL(8,2) (min)
MISSION 1 : N Satellites
  • PK Mission_ID : INT
  • Mission_Name : VARCHAR(100)
  • Mission_Type : VARCHAR(50)
  • Start_Date : DATE
  • End_Date : DATE
  • Mission_Status : VARCHAR(20)
  • Mission_Objective : TEXT
  • FK Satellite_ID : INT (→ SATELLITE)
TELEMETRY High Volume
  • PK Telemetry_ID : BIGINT
  • FK Satellite_ID : INT (→ SATELLITE)
  • Timestamp : DATETIME
  • Temperature : DECIMAL(6,2)
  • Battery_Level : DECIMAL(5,2)
  • Voltage : DECIMAL(6,2)
  • Power_Consumption : DECIMAL(8,2)
  • Signal_Strength : DECIMAL(6,2)
  • Position / Orientation : VARCHAR
GROUND_STATION Earth Segment
  • PK Station_ID : INT
  • Station_Name : VARCHAR(100)
  • Latitude : DECIMAL(9,6)
  • Longitude : DECIMAL(9,6)
  • Elevation_Meters : INT
  • Antenna_Size_Meters : DECIMAL(6,2)

  • PK Log_ID : INT
  • FK Satellite_ID : INT (→ SATELLITE)

  • Data_Received_MB : DECIMAL(10,2)
  • Status : VARCHAR(20)
